                          완료 {matchedGroup.completedCount}건

                      )}

                  <div className="text-right">
                    <div className="text-rose-600 font-bold text-lg font-mono">{formatMoney(entryDue)}</div>

                    rows={2}
                    value={entryNote}
                    onChange={(e) => setEntryNote(e.target.value)}
                    placeholder="수금 관련 메모 사항"
